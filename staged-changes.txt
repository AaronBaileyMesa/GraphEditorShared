+    private func handleControlAction(kind: ControlKind, ownerID: NodeID?) async {
+        switch kind {

+        case .redo:

+            }
+        case .toggleExpansion:
+            if let id = ownerID {
+                await toggleExpansion(for: id)
+            }
+        }

++++++++    /// Core func: Regenerates ephemerals based on selection, mode, uiConfig
++++++++    @MainActor
++++++++    public func updateEphemerals(selectedNodeID: NodeID? = nil) {
++++++++        ephemeralControlNodes.removeAll()
++++++++        ephemeralControlEdges.removeAll()
++++++++        
++++++++        // Global controls (e.g., undo/redo, configMode) – positioned near centroid
++++++++        addGlobalControls()
++++++++        
++++++++        // Per-node controls if selected
++++++++        if let ownerID = selectedNodeID ?? self.selectedNodeID {  // Assume selectedNodeID is a property; adjust if in ViewModel
++++++++            addControlsForNode(ownerID)
++++++++        }
++++++++        
++++++++        objectWillChange.send()
++++++++        changesPublisher.send()
++++++++    }

++++++++    private func addGlobalControls() {
++++++++        let centroid = physicsEngine.centroid ?? .zero
++++++++        var globalKinds: [ControlKind] = []
++++++++        
++++++++        if canUndo { globalKinds.append(.undo) }
++++++++        if canRedo { globalKinds.append(.redo) }
++++++++        globalKinds.append(.configMode)  // Always available
++++++++        
++++++++        let filtered = globalKinds.filter { kind in
++++++++            globalUiConfig.first(where: { $0.kind == kind })?.isVisible ?? true
++++++++        }.sorted { a, b in
++++++++            (globalUiConfig.first(where: { $0.kind == a })?.priority ?? 0) >
++++++++            (globalUiConfig.first(where: { $0.kind == b })?.priority ?? 0)
++++++++        }
++++++++        
++++++++        let clusterRadius: CGFloat = 20.0  // Suitable for watch
++++++++        let angleStep = CGFloat.pi * 2 / CGFloat(filtered.count)
++++++++        
++++++++        for (index, kind) in filtered.enumerated() {
++++++++            let angle = CGFloat(index) * angleStep
++++++++            let offset = CGPoint(x: cos(angle) * clusterRadius, y: sin(angle) * clusterRadius)
++++++++            let position = centroid + offset
++++++++            
++++++++            let config = globalUiConfig.first(where: { $0.kind == kind }) ?? ControlConfig(kind: kind)
++++++++            
++++++++            var control = ControlNode(
++++++++                position: position,
++++++++                ownerID: nil,
++++++++                kind: kind,
++++++++                isVisible: config.isVisible,
++++++++                priority: config.priority
++++++++            )
++++++++            
++++++++            control.action = { [weak self] in
++++++++                self?.handleControlAction(kind: kind, ownerID: nil)
++++++++            }
++++++++            
++++++++            if control.isVisible {
++++++++                ephemeralControlNodes.append(control)
++++++++            }
++++++++        }
++++++++    }

++++++++    private func addControlsForNode(_ ownerID: NodeID) {
++++++++        guard let owner = nodes.first(where: { $0.id == ownerID })?.unwrapped else { return }
++++++++        
++++++++        let isToggle = owner is ToggleNode
++++++++        var kinds: [ControlKind] = [.addNode, .addToggleNode, .delete]
++++++++        if isToggle { kinds.append(.toggle) }
++++++++        if mode == .tree { kinds.append(.addChild) }
++++++++        
++++++++        let nodeConfig = uiConfig[ownerID] ?? []
++++++++        let filtered = kinds.filter { kind in
++++++++            nodeConfig.first(where: { $0.kind == kind })?.isVisible ?? true
++++++++        }.sorted { a, b in
++++++++            (nodeConfig.first(where: { $0.kind == a })?.priority ?? 0) >
++++++++            (nodeConfig.first(where: { $0.kind == b })?.priority ?? 0)
++++++++        }
++++++++        
++++++++        let clusterRadius: CGFloat = owner.radius * 1.8
++++++++        let angleStep = CGFloat.pi * 2 / CGFloat(filtered.count)
++++++++        
++++++++        for (index, kind) in filtered.enumerated() {
++++++++            let angle = CGFloat(index) * angleStep - .pi / 2  // Start at top
++++++++            let offset = CGPoint(x: cos(angle) * clusterRadius, y: sin(angle) * clusterRadius)
++++++++            let position = owner.position + offset
++++++++            
++++++++            let config = nodeConfig.first(where: { $0.kind == kind }) ?? ControlConfig(kind: kind)
++++++++            
++++++++            var control = ControlNode(
++++++++                position: position,
++++++++                ownerID: ownerID,
++++++++                kind: kind,
++++++++                isVisible: config.isVisible,
++++++++                priority: config.priority
++++++++            )
++++++++            
++++++++            control.action = { [weak self] in
++++++++                self?.handleControlAction(kind: kind, ownerID: ownerID)
++++++++            }
++++++++            
++++++++            if control.isVisible {
++++++++                ephemeralControlNodes.append(control)
++++++++                
++++++++                // Optional spring edge (disabled in config mode)
++++++++                if !isConfigMode {
++++++++                    ephemeralControlEdges.append(GraphEdge(from: ownerID, target: control.id, type: .association))
++++++++                }
++++++++            }
++++++++        }
++++++++    }

++++++++    /// Dispatches control tap actions
++++++++    private func handleControlAction(kind: ControlKind, ownerID: NodeID?) {
++++++++        pushUndo()  // Snapshot before action
++++++++        
++++++++        switch kind {
++++++++        case .addNode:
++++++++            // Implement or call existing addNode(at:)
++++++++            let position = ownerID.flatMap { nodes.first(where: { $0.id == $0 })?.position } ?? .zero
++++++++            addNode(at: position + CGPoint(x: 20, y: 20))  // Near owner if present
++++++++        case .addToggleNode:
++++++++            // Similar to above
++++++++            let position = ownerID.flatMap { nodes.first(where: { $0.id == $0 })?.position } ?? .zero
++++++++            addToggleNode(at: position + CGPoint(x: 20, y: 20))
++++++++        case .delete:
++++++++            if let id = ownerID { deleteNode(withID: id) }
++++++++        case .toggle:
++++++++            if let id = ownerID { toggleExpansion(for: id) }
++++++++        case .undo:
++++++++            undo()
++++++++        case .redo:
++++++++            redo()
++++++++        case .addChild:
++++++++            if let id = ownerID { addChildToNode(id) }
++++++++        case .configMode:
++++++++            isConfigMode.toggle()
++++++++        }
++++++++        
++++++++        #if os(watchOS)
++++++++        WKInterfaceDevice.current().play(.click)
++++++++        #endif
++++++++    }

++++++++    /// Toggle visibility (persisted)
++++++++    public func toggleControlVisibility(kind: ControlKind, ownerID: NodeID?) {
++++++++        if let ownerID = ownerID {
++++++++            var configs = _uiConfig[ownerID] ?? []
++++++++            if let index = configs.firstIndex(where: { $0.kind == kind }) {
++++++++                configs[index].isVisible.toggle()
++++++++            } else {
++++++++                configs.append(ControlConfig(kind: kind, isVisible: false))
++++++++            }
++++++++            _uiConfig[ownerID] = configs
++++++++        } else {
++++++++            if let index = _globalUiConfig.firstIndex(where: { $0.kind == kind }) {
++++++++                _globalUiConfig[index].isVisible.toggle()
++++++++            } else {
++++++++                _globalUiConfig.append(ControlConfig(kind: kind, isVisible: false))
++++++++            }
++++++++        }
++++++++        updateEphemerals()
++++++++        pushUndo()
++++++++    }
